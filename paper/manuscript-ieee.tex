\documentclass[journal]{IEEEtran}

% IEEE requires Times. IEEEtran asks for `ptm', which bare XeTeX cannot
% resolve, so load TeX Gyre Termes --- the metric-compatible Times clone ---
% explicitly for both text and math.
%
% Building this file emits a handful of "Font shape `TU/ptm/...' undefined"
% warnings during class loading. They are harmless: they come from IEEEtran.cls
% declaring its own defaults before this preamble runs, and \setmainfont wins
% afterwards. Verify with `pdffonts manuscript-ieee.pdf` --- the embedded faces
% should all be TeXGyreTermes.
\usepackage{fontspec}
\setmainfont{texgyretermes}[
  Extension      = .otf,
  UprightFont    = *-regular,
  BoldFont       = *-bold,
  ItalicFont     = *-italic,
  BoldItalicFont = *-bolditalic ]

\input{preamble-common}

% unicode-math must follow amsmath (loaded in preamble-common). Without it the
% text sets in Times while every digit inside $...$ sets in Computer Modern,
% which is conspicuous in tables built from figures like $1.0000 \pm 0.0000$.
\usepackage{unicode-math}
\setmathfont{texgyretermes-math.otf}

\usepackage{stfloats}              % permit full-width floats at page bottoms
\usepackage[hidelinks]{hyperref}   % load late
\usepackage{xurl}                  % after hyperref: let DOIs/URLs break anywhere

% Two columns, and twelve of the thirteen tables are wider than one column.
% \wtable spans both columns; \ntable is the single-column case.
\newenvironment{wtable}{\begin{table*}[!t]}{\end{table*}}
\newenvironment{ntable}{\begin{table}[!t]}{\end{table}}
\newenvironment{wfigure}{\begin{figure*}[!t]}{\end{figure*}}

% A table-heavy paper starves the float queue at IEEEtran's defaults, which
% pushes tables many pages past the text that discusses them. Loosen the
% placement parameters so each table lands near its argument.
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.75}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.70}
